5. Какой тип доступа называется private?
Тип доступа private (закрытый / собственный) означает, что член-данные и методы класса скрыты от внешнего использования и доступны только внутри методов самого этого класса, а также дружественным (friend) функциям и классам.
ПРАКТИКА БИЛЕТ 1: Шаблон Stack (все методы вне класса)
#include <iostream>
using namespace std;

template <class T>
class Stack 
{
    T* arr;
    int max;
    int n;

public:
    // а) конструкторы (все)
    Stack(int k = 50);
    Stack(const Stack<T>& s);

    // б) деструктор
    ~Stack();

    // в) проверка на пустоту
    bool IsEmpty() const;

    // г) добавить в стек элемент
    bool Push(const T& elem);

    // д) перегрузить оператор потокового вывода
    template <class U>
    friend ostream& operator<<(ostream& out, const Stack<U>& s);
};

// Определения методов ВНЕ класса
template <class T>
Stack<T>::Stack(int k) 
{
    max = (k > 0) ? k : 50;
    n = 0;
    arr = new T[max];
}

template <class T>
Stack<T>::Stack(const Stack<T>& s) 
{
    max = s.max;
    n = s.n;
    arr = new T[max];
    for (int i = 0; i < n; i++) {
        arr[i] = s.arr[i];
    }
}

template <class T>
Stack<T>::~Stack() 
{
    delete[] arr;
}

template <class T>
bool Stack<T>::IsEmpty() const 
{
    return n == 0;
}

template <class T>
bool Stack<T>::Push(const T& elem) 
{
    if (n == max) return false;
    arr[n++] = elem;
    return true;
}

template <class U>
ostream& operator<<(ostream& out, const Stack<U>& s) 
{
    for (int i = 0; i < s.n; i++) {
        out << s.arr[i] << " ";
    }
    return out;
}

5. Что такое конструктор?
Конструктор — это специальный метод класса, который вызывается автоматически при создании объекта и предназначен для инициализации его член-данных (и выделения ресурсов). Его имя совпадает с именем класса, и он не имеет возвращаемого значения.
ПРАКТИКА БИЛЕТ 4: Класс Array (с FindKey)
#include <iostream>
using namespace std;

class Array {
private:
    int* arr;
    int n;

public:
    // 1. Конструкторы и деструктор
    Array(int n_ = 0);
    Array(const int* a, int n_);
    Array(const Array& other);
    ~Array();

    // 2. Функция поиска ключа (возвращает индекс или -1)
    int FindKey(int key) const;

    // 3. Перегрузка операторов
    Array& operator=(const Array& other);
    int& operator[](int index);

    Array& operator+=(int key);     // добавление ключа в конец (меняет текущий)
    Array operator+(int key) const; // добавление ключа в конец (создает новый)

    Array& operator-=(int key);     // удаление 1-го вхождения key (меняет текущий)
    Array operator-(int key) const; // удаление 1-го вхождения key (создает новый)

    bool operator==(const Array& other) const;
    bool operator!=(const Array& other) const;

    friend istream& operator>>(istream& is, Array& a);
    friend ostream& operator<<(ostream& os, const Array& a);
};

// Конструктор по умолчанию / с размером
Array::Array(int n_) {
    n = (n_ > 0) ? n_ : 0;
    if (n > 0) {
        arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = 0;
    } else arr = nullptr;
}

// Конструктор перевода обычного массива в объект Array
Array::Array(const int* a, int n_) {
    n = (n_ > 0) ? n_ : 0;
    if (n > 0) {
        arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = a[i];
    } else arr = nullptr;
}

// Конструктор копирования
Array::Array(const Array& other) {
    n = other.n;
    if (n > 0) {
        arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = other.arr[i];
    } else arr = nullptr;
}

// Деструктор
Array::~Array() {
    delete[] arr;
}

// ФУНКЦИЯ ПОИСКА КЛЮЧА: возвращает индекс или -1
int Array::FindKey(int key) const {
    for (int i = 0; i < n; i++) {
        if (arr[i] == key) return i;
    }
    return -1;
}

// Оператор присвоения (=)
Array& Array::operator=(const Array& other) {
    if (this != &other) { // Проверка на самоприсваивание
        delete[] arr;     // Освобождаем старую память
        n = other.n;
        if (n > 0) {
            arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = other.arr[i];
        } else arr = nullptr;
    }
    return *this;
}

// Оператор индексации ([]) — возвращает ссылку
int& Array::operator[](int index) {
    return arr[index];
}

// Оператор += (добавление ключа key в конец массива)
Array& Array::operator+=(int key) {
    int* temp = new int[n + 1];
    for (int i = 0; i < n; i++) temp[i] = arr[i];
    temp[n] = key;
    delete[] arr;
    arr = temp;
    n++;
    return *this;
}

// Оператор + (добавление ключа key в конец, новый объект)
Array Array::operator+(int key) const {
    Array res(*this);
    res += key;
    return res;
}

// Оператор -= (удаление первого вхождения key через FindKey)
Array& Array::operator-=(int key) {
    int pos = FindKey(key);
    if (pos == -1) return *this;

    if (n == 1) {
        delete[] arr;
        arr = nullptr;
        n = 0;
        return *this;
    }

    int* temp = new int[n - 1];
    int j = 0;
    for (int i = 0; i < n; i++) {
        if (i != pos) temp[j++] = arr[i];
    }
    delete[] arr;
    arr = temp;
    n--;
    return *this;
}

// Оператор - (удаление первого вхождения key, новый объект)
Array Array::operator-(int key) const {
    Array res(*this);
    res -= key;
    return res;
}

// Операторы сравнения (== и !=)
bool Array::operator==(const Array& other) const {
    if (n != other.n) return false;
    for (int i = 0; i < n; i++) {
        if (arr[i] != other.arr[i]) return false;
    }
    return true;
}

bool Array::operator!=(const Array& other) const {
    return !(*this == other);
}

// Потоковый ввод (>>) и вывод (<<)
istream& operator>>(istream& is, Array& a) {
    for (int i = 0; i < a.n; i++) is >> a.arr[i];
    return is;
}

ostream& operator<<(ostream& os, const Array& a) {
    for (int i = 0; i < a.n; i++) os << a.arr[i] << " ";
    return os;
}

Неверно. Объявление дружественной функции с ключевым словом friend можно помещать в любом месте внутри класса (на него не влияют спецификаторы доступа private, protected, public).
3. Статические методы не могут вызываться через объекты, как обычные методы.
Неверно. Статические методы можно вызывать как через имя класса (Class::Method()), так и через объект класса (obj.Method()).

5. Что такое деструктор?
Деструктор — это специальный метод класса, который вызывается автоматически при удалении (уничтожении) объекта для освобождения динамической памяти и ресурсов. Имеет имя ~Имя_класса, не имеет параметров и возвращаемого типа.

Билет 21 (Практика FreeVector)
Обязательно написать в бланке: Конструктор копирования и деструктор не требуются, т.к. в классе FreeVector нет указателей и не выделяется динамическая память (все поля double x, y, z лежат в стеке). Стандартный конструктор копирования (побитовое копирование) и деструктор по умолчанию работают корректно. 
#include <iostream>
using namespace std;

class FreeVector {
private:
    double x, y, z;

public:
    // Конструктор с параметрами по умолчанию (заменяет оба конструктора)

    }

{
    if (n <= 0) return -1;
    int minInd = 0;
    for (int i = 1; i < n; i++) {
        if (arr[i] < arr[minInd]) {
            minInd = i;
        }
    }
    return minInd;
}

    }

}

        if (i != pos) {
            temp[j++] = arr[i];
        }
    }

    arr = temp;
    n--;
    return *this;
}

#include <iostream>
#include <cstring>
using namespace std;

class String {
private:
    char* line; // указатель на строку
    int len;    // длина строки

public:
    String(int L = 80);               // конструктор по умолчанию / заданной длины
    String(const char* s);            // конструктор инициализации C-строкой
    String(const String& other);      // конструктор копирования
    ~String();                        // деструктор

    String& operator=(const String& other);      // = (присвоение)
    char& operator[](int index);                 // [] (возвращает ссылку на символ)

    String operator+(const String& other) const; // + (конкатенация — создает новую строку)
    String& operator+=(const String& other);     // += (конкатенация — изменяет текущую строку)

    bool operator==(const String& other) const;  // == (сравнение строк на равенство)
    bool operator!=(const String& other) const;  // != (сравнение строк на неравенство)

    friend istream& operator>>(istream& in, String& s);
    friend ostream& operator<<(ostream& out, const String& s);
};

// ================= ОПРЕДЕЛЕНИЕ МЕТОДОВ ВНЕ КЛАССА =================

String::String(int L) {
    len = (L > 0) ? L : 0;
    line = new char[len + 1];
    line[0] = '\0';
}

String::String(const char* s) {
    len = strlen(s);
    line = new char[len + 1];
    strcpy(line, s);
}

String::String(const String& other) {
    len = other.len;
    line = new char[len + 1];
    strcpy(line, other.line);
}

String::~String() {
    delete[] line;
}

String& String::operator=(const String& other) {
    if (this != &other) {
        delete[] line;
        len = other.len;
        line = new char[len + 1];
        strcpy(line, other.line);
    }
    return *this;
}

char& String::operator[](int index) {
    return line[index];
}

String String::operator+(const String& other) const {
    String res(len + other.len);
    strcpy(res.line, line);
    strcat(res.line, other.line);
    return res;
}

String& String::operator+=(const String& other) {
    len += other.len;
    char* temp = new char[len + 1];
    strcpy(temp, line);
    strcat(temp, other.line);
    delete[] line;
    line = temp;
    return *this;
}

bool String::operator==(const String& other) const {
    return strcmp(line, other.line) == 0;
}

bool String::operator!=(const String& other) const {
    return strcmp(line, other.line) != 0;
}

istream& operator>>(istream& in, String& s) {
    char buffer[256];
    in >> buffer;
    delete[] s.line;
    s.len = strlen(buffer);
    s.line = new char[s.len + 1];
    strcpy(s.line, buffer);
    return in;
}

ostream& operator<<(ostream& out, const String& s) {
    out << s.line;
    return out;
}
6. Stack<T> (Шаблон стека) 
#include <iostream>
using namespace std;

template <class T>
class Stack {
private:
    T* arr;  // массив элементов стека
    int max; // максимальный размер массива arr
    int n;   // количество реальных элементов в стеке

public:
    Stack(int k = 50);                  // конструктор (по умолчанию k = 50)
    Stack(const Stack<T>& other);       // конструктор копирования
    ~Stack();                           // деструктор

    bool IsEmpty() const;               // проверка на пустоту
    bool IsFull() const;                // проверка на заполненность
    bool Push(const T& item);           // добавить в стек элемент
    bool Pop(T& item);                  // удалить из стека элемент
    T Top() const;                      // взять значение с вершины стека

    Stack<T>& operator=(const Stack<T>& other); // = (присвоение)

    template <class U>
    friend ostream& operator<<(ostream& out, const Stack<U>& s);
};

// ================= ОПРЕДЕЛЕНИЕ МЕТОДОВ ШАБЛОНА ВНЕ КЛАССА =================

template <class T>
Stack<T>::Stack(int k) {
    max = (k > 0) ? k : 50;
    n = 0;
    arr = new T[max];
}

template <class T>
Stack<T>::Stack(const Stack<T>& other) {
    max = other.max;
    n = other.n;
    arr = new T[max];
    for (int i = 0; i < n; i++) {
        arr[i] = other.arr[i];
    }
}

template <class T>
Stack<T>::~Stack() {
    delete[] arr;
}

template <class T>
bool Stack<T>::IsEmpty() const {
    return n == 0;
}

template <class T>
bool Stack<T>::IsFull() const {
    return n == max;
}

template <class T>
bool Stack<T>::Push(const T& item) {
    if (IsFull()) return false;
    arr[n++] = item;
    return true;
}

template <class T>
bool Stack<T>::Pop(T& item) {
    if (IsEmpty()) return false;
    item = arr[--n];
    return true;
}

template <class T>
T Stack<T>::Top() const {
    return arr[n - 1];
}

template <class T>
Stack<T>& Stack<T>::operator=(const Stack<T>& other) {
    if (this != &other) {
        delete[] arr;
        max = other.max;
        n = other.n;
        arr = new T[max];
        for (int i = 0; i < n; i++) {
            arr[i] = other.arr[i];
        }
    }
    return *this;
}

template <class U>
ostream& operator<<(ostream& out, const Stack<U>& s) {
    for (int i = 0; i < s.n; i++) {
        out << s.arr[i] << " ";
    }
    return out;
}
📖 Теоретическая часть (15 вопросов)
1. Принципы ООП, класс, this
Принципы ООП:
  1. Инкапсуляция — объединение данных и методов их обработки в классе + защита данных от прямого доступа извне (сокрытие).
  2. Наследование — создание нового класса (потомка) на базе существующего с перенятием его свойств и методов.
  3. Полиморфизм — один интерфейс и множество реализаций (перегрузка функций/операций, виртуальные методы).
Класс — пользовательский тип данных, объединяющий член-данные и член-функции (методы).
Объект — конкретный экземпляр (переменная) типа класс в памяти.
Сокрытие информации — размещение данных в секции private, доступ извне — только через public-методы.
Отличие методов от обычных функций: объявляются внутри класса; имеют доступ к private-членам; вызываются через объект (. или ->); имеют неявный указатель this.
Типы доступа:
  • private (закрытый, по умолчанию) — доступ только методам самого класса и friend.
  • protected (защищенный) — доступ методам самого класса, friend и классам-наследникам.
  • public (открытый) — доступ из любого места программы (в т.ч. в main).
Указатель this — константный указатель (Class* const this), который неявно передается в любой нестатический метод и хранит адрес объекта, для которого вызван метод.
2. Методы, операция ::, внутри/вне
Отличие член-функций от обычных: объявлены внутри класса; имеют доступ к private-полям; вызываются через объект; содержат указатель this.
Операция :: (разрешение области видимости) — указывает, какому классу принадлежит метод или статическое поле при определении вне класса (Class::Method()).
Внутри класса vs Вне класса:
  • Определенные внутри класса автоматически считаются встраиваемыми (inline) — компилятор подставляет их код в место вызова.
  • Определенные вне класса (через ::) компилируются как обычные функции (если явно не написать inline).
3. Конструкторы и деструкторы
Конструктор — спец. метод для автоматической инициализации член-данных объекта при его создании (и выделения памяти).
Особенности конструктора: имя совпадает с именем класса; нет возвращаемого значения (даже void); вызывается автоматически; может иметь аргументы и быть перегружен.
Деструктор — спец. метод для освобождения ресурсов (динамической памяти) перед уничтожением объекта.
Особенности деструктора: имя ~ИмяКласса(); нет возвращаемого типа; всегда без аргументов; не может быть перегружен; вызывается автоматически при выходе из области видимости или при delete.
4. Конструктор копирования vs =
Конструктор копирования A(const A& other) — создает новый объект, копируя данные из существующего объекта того же класса.
Когда необходим и почему: обязателен, если член-данные находятся в динамической памяти (есть указатели). Иначе конструктор по умолчанию сделает поверхностное копирование (два указателя на одну память → порча данных и двойной delete[]).
Отличие от операции =:
  1. Конструктор копирования создает новый объект, а = изменяет уже существующий.
  2. В = нужна проверка this != &other и очистка старой памяти delete[].
  3. Операция = возвращает *this по ссылке.
5. Правила перегрузки. + и +=
Правила перегрузки операций:
  1. Хотя бы один операнд должен быть пользовательского типа (класса).
  2. Нельзя создавать новые знаки операций (например, ~=).
  3. Нельзя менять приоритет, ассоциативность и местность (арность).
  4. Нельзя перегружать: ., .*, ::, ?:, sizeof.
  5. Операции =, [], (), -> перегружаются только как методы класса.
  6. Как метод класса: унарная — 0 аргументов, бинарная — 1 аргумент. Как friend: унарная — 1 аргумент, бинарная — 2 аргумента.
Отличие + от +=: + не меняет операнды, создает новый объект и возвращает его по значению; += изменяет левый операнд (*this) и возвращает его по ссылке.
6. Перегрузка операций [] и ()
Индексация []: перегружается только как нестатический метод класса; имеет 1 аргумент (индекс); возвращает ссылку на элемент (T& operator[](int i)), чтобы объект мог стоять слева от =.
Вызов функции (): перегружается только как нестатический метод класса; единственный оператор, который может иметь произвольное число аргументов (от 0 и более).
7. Перегрузка (тип) и ++
Приведение типа operator тип(): только как метод класса; без аргументов; перед словом operator возвращаемый тип не пишется.
Инкремент ++:
  • Префиксный (++a): без аргументов A& operator++(), меняет *this и возвращает ссылку *this.
  • Постфиксный (a++): с фиктивным параметром int A operator++(int), сохраняет копию, меняет *this, возвращает старую копию по значению.
8. Перегрузка операции =
Перегружается только как нестатический метод класса. Обязательна в классах с динамической памятью.
Шаги внутри operator=: 1) проверка на самоприсваивание if (this == &other) return *this;; 2) освобождение старой памяти delete[]; 3) выделение новой памяти и копирование; 4) return *this;.
9. Дружественность, потоки << и >>
Другом класса (friend) может быть: обычная функция, метод другого класса, другой класс целиком. Имеет доступ к private-членам, не имеет this.
Общее у << и >>: перегружаются как friend с 2 аргументами; 1-й аргумент и возврат — ссылка на поток.
Разница: в << поток ostream& и объект передается по const-ссылке; в >> поток istream& и объект по неконстантной ссылке.
Почему нельзя как метод класса: в методе класса левым операндом всегда идет объект класса (через this), вызов был бы obj << cout.
Обязательна ли ссылка: да, потоки запрещено копировать + ссылка нужна для цепочек и записи в объект.
10. Массивы объектов
Статический массив: без списка инициализации — вызывается конструктор по умолчанию; со списком инициализации { A(1), A(2,3) } — можно вызывать любые (в т.ч. разные) конструкторы.
Динамический массив (new A[n]): явно инициализировать параметрами нельзя, вызывается только конструктор по умолчанию.
Порядок вызова: конструкторы — от 0 до n-1; деструкторы — в обратном порядке от n-1 до 0.
11-12. Агрегированные классы
Агрегированный класс — класс, содержащий в качестве член-данных объекты или указатели на объекты другого (используемого) класса.
Если поле — указатель (A* ptr;): в конструкторе включающего класса нужно явно выделить память через ptr = new A(аргументы);.
Как получить доступ к private-полям используемого класса: 1) через его открытые (public) методы; 2) объявить агрегированный класс другом (friend) внутри используемого класса.
13. Функции- и классы-шаблоны
Шаблон — параметризованное описание функции или класса, где типы данных, аргументов и возвращаемых значений методов заданы параметром template <class T>.
Порожденная функция — конкретная функция, сгенерированная компилятором из шаблона при подстановке конкретного типа вместо T.
Использование: для функции тип выводится автоматически по аргументам; для класса-шаблона тип указывается явно в угловых скобках: Stack<int> s;.
14. Статические член-данные
Объявляются внутри класса со словом static. Существуют в единственном экземпляре на весь класс (общие для всех объектов) даже без создания объектов.
Инициализация: обязательно вне класса через :: без слова static (int A::count = 0;). Тип доступа (private/public) на инициализацию не влияет.
В main: если поле public — доступно и изменяемо через A::count или obj.count; если private — доступ только через public-методы.
15. Статические методы класса
Объявляются в классе со словом static, определяются вне класса через :: без слова static.
Особенности: 1) не имеют указателя this; 2) могут обращаться напрямую только к статическим членам класса; 3) если метод public, вызывается в main через Class::Method() или obj.Method() (если private — вызвать в main нельзя).
